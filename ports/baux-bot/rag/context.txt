get_directory_access() {
    local dir_path="$1"
    for config in "${DIRECTORY_CONFIG[@]}"; do
        local config_dir config_access config_desc
        IFS=':' read -r config_dir config_access config_desc <<< "$config"
        if [[ "$dir_path" == "$config_dir" ]]; then
            echo "$config_access"
            return
        fi
    done
    echo "unknown"
}

is_readonly() {
    local dir_path="$1"
    # Check if the directory or any parent directory is readonly
    while [[ "$dir_path" != "/" ]]; do
        if [[ "$(get_directory_access "$dir_path")" == "readonly" ]]; then
            return 0
        fi
        dir_path=$(dirname "$dir_path")
    done
    return 1

# BAUXWM status bar — shows session info when BAUXWM=1

get_sessions() {
    # Get tmux sessions if available

        tmux list-sessions 2>/dev/null | awk -F: '{print $1}' | tr '\n' ' ' | sed 's/ $//'
    else
        echo ""
    fi
}

    # Try to get battery info (FreeBSD)

        echo "AC"
    else
        echo ""
    fi

# Roxanne - Debian Fork of RoxieOS
**RoxieOS is FreeBSD based, this is a fork for Debian Development**

This directory contains the complete Debian-based fork of RoxieOS, enhanced with the most successful concepts from both FreeBSD and Debian development paths. The Debian fork excels at accessibility and live system builds, while FreeBSD leads in mesh networking and AI integration. It exists mostly for rapid development of consepts when tools aren't avaliable in FreeBSD (such as access opencode). It is intened to be an "assist" with development only, not as the primary project.

## Core Philosophy: Best of Both Worlds

**Debian Strengths (Preserved Here):**
- Comprehensive accessibility font stack (9 professional fonts)
- Complete live system build with debootstrap
- Working local apt repository infrastructure
- Custom kernel builds and bootloader themes

**FreeBSD Strengths (Integrated Concepts):**
- Headscale mesh networking for distributed sessions
- AI-powered development (xai-chat, baux-bot)
- Production-ready installation scripts
- Advanced session persistence and resurrection
